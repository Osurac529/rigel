// C++ Reference Implementation
struct Vertex {
    vec3 position;
    vec2 uv;
    vec3 normal;
    vec3 tangent;
    vec3 bitangent;
};

void calculateTBN(Vertex& v1, Vertex& v2, Vertex& v3) {
    vec3 edge1 = v2.position - v1.position;
    vec3 edge2 = v3.position - v1.position;
    vec2 deltaUV1 = v2.uv - v1.uv;
    vec2 deltaUV2 = v3.uv - v1.uv;

    float f = 1.0f / (deltaUV1.x * deltaUV2.y - deltaUV2.x * deltaUV1.y);

    vec3 tangent;
    tangent.x = f * (deltaUV2.y * edge1.x - deltaUV1.y * edge2.x);
    tangent.y = f * (deltaUV2.y * edge1.y - deltaUV1.y * edge2.y);
    tangent.z = f * (deltaUV2.y * edge1.z - deltaUV1.y * edge2.z);

    vec3 bitangent;
    bitangent.x = f * (-deltaUV2.x * edge1.x + deltaUV1.x * edge2.x);
    bitangent.y = f * (-deltaUV2.x * edge1.y + deltaUV1.x * edge2.y);
    bitangent.z = f * (-deltaUV2.x * edge1.z + deltaUV1.x * edge2.z);

    // Accumulate to vertices (average shared vertex tangents later)
    v1.tangent += tangent; v2.tangent += tangent; v3.tangent += tangent;
    v1.bitangent += bitangent; v2.bitangent += bitangent; v3.bitangent += bitangent;
}

// Fragment Shader
in vec3 fragNormal;
in vec4 fragTangent;

void main() {
    vec3 N = normalize(fragNormal);
    vec3 T = normalize(fragTangent.xyz);
    
    // Gram-Schmidt re-orthogonalization
    T = normalize(T - dot(T, N) * N);
    
    // Reconstruct the bitangent, multiplying by the handedness sign!
    vec3 B = cross(N, T) * fragTangent.w;
    
    mat3 TBN = mat3(T, B, N);
    
    // Now you can sample your normal map safely!
}

layout (location = 0) in vec3 inPosition;
layout (location = 1) in vec2 inUV;
layout (location = 2) in vec3 inNormal;
layout (location = 3) in vec3 inTangent;

uniform mat4 model;
uniform mat3 normalMatrix; // Inverse transpose of model matrix

out vec2 fragUV;
out mat3 outTBN;

void main() {
    fragUV = inUV;
    
    // Transform vectors to world space
    vec3 N = normalize(normalMatrix * inNormal);
    vec3 T = normalize(normalMatrix * inTangent);
    
    // Gram-Schmidt re-orthogonalization
    T = normalize(T - dot(T, N) * N);
    
    // Reconstruct Bitangent via cross product
    vec3 B = cross(N, T);
    
    // Construct TBN matrix mapping tangent -> world space
    outTBN = mat3(T, B, N);
    
    gl_Position = model * vec4(inPosition, 1.0);
}
